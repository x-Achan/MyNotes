
创建 springboot项目
创建数据库，插入表，插入数据
填写数据库配置文件 yaml
```yaml
# 配置 Spring Boot 应用的名称
spring:  
  application:  
    name: tlias-web-mangement   # 项目名
  #数据库的连接信息，SpringBoot根据这里的信息创建数据库连接池DataSource对象
  datasource:  
	url: jdbc:mysql://localhost:3306/tlias  # 数据库连接地址
    driver-class-name: com.mysql.cj.jdbc.Driver   # 数据库驱动
    username: root  
    password: 123456  
  
#Mybatis的相关配置  
mybatis:  
  configuration:  
	# 配置 MyBatis 日志实现，打印SQL
    log-impl: org.apache.ibatis.logging.stdout.StdOutImpl  
  mapper-locations: classpath:mapper/*.xml
```
